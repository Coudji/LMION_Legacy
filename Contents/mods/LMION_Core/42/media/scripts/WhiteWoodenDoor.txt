-- Shared native definition for the vanilla white wooden door family.
-- The template owns the physical SpriteConfig once; feature modules can
-- rebuild Base.WhiteWoodenDoor from it without duplicating sprite ownership.

module Base
{
    template entity WhiteWoodenDoorTemplate
    {
        component SpriteConfig
        {
            skillBaseHealth = 300,
            BreakSound = BreakDoor,

            face W
            {
                layer
                {
                    row = fixtures_doors_01_0,
                }
            }

            face N
            {
                layer
                {
                    row = fixtures_doors_01_1,
                }
            }

            face W_OPEN
            {
                layer
                {
                    row = fixtures_doors_01_2,
                }
            }

            face N_OPEN
            {
                layer
                {
                    row = fixtures_doors_01_3,
                }
            }
        }
    }

    entity WhiteWoodenDoor
    {
        entitytemplate = WhiteWoodenDoorTemplate,
    }
}
